restart_vm.sh
#!/usr/bin/env bash

set -euo pipefail

VM_NAME="$1"

if [[ -z "${VM_NAME}" ]]; then
  echo "Usage: $0 <vm-name>"
  exit 1
fi

get_status() {
  yc compute instance get --name "${VM_NAME}" --format json | jq -r '.status'
}

echo "[*] Checking current status..."
STATUS="$(get_status)"
echo "    status: ${STATUS}"

if [[ "${STATUS}" == "RUNNING" ]]; then
  echo "[*] Stopping VM: ${VM_NAME}"
  yc compute instance stop --name "${VM_NAME}"

  echo "[*] Waiting for VM to stop..."
  while true; do
    STATUS="$(get_status)"
    echo "    status: ${STATUS}"
    [[ "${STATUS}" == "STOPPED" ]] && break
    sleep 3
  done
else
  echo "[*] VM is not running, skipping stop."
fi

echo "[*] Starting VM: ${VM_NAME}"
yc compute instance start --name "${VM_NAME}"

echo "[*] Waiting for VM to start..."
while true; do
  STATUS="$(get_status)"
  echo "    status: ${STATUS}"
  [[ "${STATUS}" == "RUNNING" ]] && break
  sleep 3
done

IP=$(yc compute instance get --name "${VM_NAME}" --format json \
  | jq -r '.network_interfaces[0].primary_v4_address.one_to_one_nat.address')

echo "[+] External IP: ${IP}"
