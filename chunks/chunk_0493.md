import sqlite3
from telegram import Update
from telegram.ext import Application, CommandHandler, ContextTypes

# Конфигурация: токен бота и путь к базе данных SQLite
TOKEN = "YOUR_TELEGRAM_BOT_TOKEN"
DB_PATH = "database.db"


def init_db():
    """Создает тестовую таблицу, если она не существует."""
    with sqlite3.connect(DB_PATH) as conn:
        cursor = conn.cursor()
        cursor.execute(
            """
            CREATE TABLE IF NOT EXISTS items (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                status TEXT NOT NULL
            )
        """
        )
        # Добавим тестовую запись для проверки, если таблица пуста
        cursor.execute("SELECT COUNT(*) FROM items")
        if cursor.fetchone()[0] == 0:
            cursor.execute(
                "INSERT INTO items (name, status) VALUES ('Сервер 1', 'Активен')"
            )
        conn.commit()


async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Отправляет приветственное сообщение."""
    await update.message.reply_text(
        "Привет! Используй команду /get [название] для поиска данных в БД.\n"
        "Пример: /get Сервер 1"
    )


async def get_data(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Ищет данные в БД по названию и возвращает результат."""
    if not context.args:
        await update.message.reply_text("Пожалуйста, укажите название. Пример: /get Сервер 1")
        return

    # Собираем аргументы команды в одну строку запроса
    search_query = " ".join(context.args)

    with sqlite3.connect(DB_PATH) as conn:
        cursor = conn.cursor()
        cursor.execute(
            "SELECT name, status FROM items WHERE name LIKE ?",
            (f"%{search_query}%",),
        )
        results = cursor.fetchall()

    if not results:
        await update.message.reply_text(f"По запросу '{search_query}' ничего не найдено.")
        return

    # Формируем ответ
    response = "Результаты поиска:\n"
    for name, status in results:
        response += f"🔹 {name} — Статус: {status}\n"

    await update.message.reply_text(response)


def main():
    """Запуск бота."""
    init_db()

    # Создаем приложение бота (используется python-telegram-bot v20+)
    app = Application.builder().token(TOKEN).build()

    # Регистрируем обработчики команд
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("get", get_data))

    print("Бот запущен и мониторит запросы...")
    app.run_polling()


if __name__ == "__main__":
    main()
