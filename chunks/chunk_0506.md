Шаг 1. Полный разгон ядра Linux + Активация BBR
Выполни этот объединенный блок команд одной вставкой в терминал каждого заграничного сервера.

# --- НАЧАЛО БЛОКА ТЮНИНГА ЯДРА (ЗАГРАНИЦА) ---
# 1. Активация алгоритма BBR (критично для стабильной отдачи пакетов)
echo "net.core.default_qdisc=fq" | sudo tee -a /etc/sysctl.conf
echo "net.ipv4.tcp_congestion_control=bbr" | sudo tee -a /etc/sysctl.conf
# 2. Максимальное увеличение буферов памяти (для TCP и UDP/Hysteria)
echo "net.core.rmem_max=16777216" | sudo tee -a /etc/sysctl.conf
echo "net.core.wmem_max=16777216" | sudo tee -a /etc/sysctl.conf
echo "net.ipv4.tcp_rmem=4096 87380 16777216" | sudo tee -a /etc/sysctl.conf
echo "net.ipv4.tcp_wmem=4096 65536 16777216" | sudo tee -a /etc/sysctl.conf
echo "net.core.netdev_max_backlog=10000" | sudo tee -a /etc/sysctl.conf
# 3. Разгон входящих очередей для приема плотного потока запросов от РФ-сервера
echo "net.core.somaxconn=16384" | sudo tee -a /etc/sysctl.conf
echo "net.ipv4.tcp_tw_reuse=1" | sudo tee -a /etc/sysctl.conf
# Применение настроек ядра Linux без перезагрузки системы
sudo sysctl -p
# --- КОНЕЦ БЛОКА ТЮНИНГА ЯДРА (ЗАГРАНИЦА) ---



Шаг 2. Контрольная проверка заграничного сервера

# 1. Проверка активных параметров ядра (скопируй одной строкой)
sysctl net.core.rmem_max net.core.wmem_max net.core.netdev_max_backlog net.core.somaxconn net.ipv4.tcp_tw_reuse
* Все выведенные цифры должны строго соответствовать значениям из Шага 1.

# 2. Проверка диапазонов буферов TCP
sysctl net.ipv4.tcp_rmem net.ipv4.tcp_wmem
* В конце каждой строки должно быть число 16777216.
